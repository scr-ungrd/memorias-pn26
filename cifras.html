Cifras de la plataforma

Cifras de participación

La PN26 se desarrolló mediante un proceso amplio de convocatoria y articulación de actores de todos los territorios y sectores del país. Se promovió la participación de actores nacionales, departamentales y municipales, entidades públicas y privadas, academia, sociedad civil, comunidades organizadas, pueblos y organizaciones étnicas, así como redes de jóvenes, mujeres y otros grupos poblacionales, bajo enfoques de género, interseccionalidad, discapacidad y diversidad étnica.

De esta manera, la PN26 se configuró como un ejercicio de construcción colectiva que articuló conocimientos técnicos y científicos con saberes locales y ancestrales, conectando las perspectivas territoriales, culturales y sectoriales con las prioridades nacionales y las agendas internacionales de reducción del riesgo de desastres.

max_dep_sqrt = Math.sqrt(d3.max(asis_dep_raw, d => d.asistentes))
max_mun_sqrt = Math.sqrt(d3.max(asis_mun_raw, d => d.asistentes))

color_dep = f => {

}

color_mun_fn = f => {

  const v = asis_mun_index.get(key) ?? 0
  return v === 0 ? ramp_teal[0] : escala_magnitud(Math.sqrt(v) / max_mun_sqrt)
}

// Función reutilizable: gráfico de pastel (dona) con leyenda y tooltip.
dibujar_donut = (data, colores, { width = 240, height = 240 } = {}) => {
  const radio = Math.min(width, height) / 2 - 8
  const total = d3.sum(data, d => d.cantidad)

  const generar_pie = d3.pie().value(d => d.cantidad).sort(null).padAngle(0.014)
  const arco = d3.arc().innerRadius(radio * 0.58).outerRadius(radio).cornerRadius(3)
  const arco_etiqueta = d3.arc().innerRadius(radio * 0.79).outerRadius(radio * 0.79)
  const arcos = generar_pie(data)

  const container = d3.create("div")
    .style("display", "flex")
    .style("flex-wrap", "wrap")
    .style("align-items", "flex-start")
    .style("gap", "16px")
    .style("font-family", "system-ui, sans-serif")

  const svgWrap = container.append("div")
    .style("position", "relative")
    .style("flex", "0 0 auto")

  const svg = svgWrap.append("svg")
    .attr("width", width)
    .attr("height", height)
    .attr("viewBox", `0 0 ${width} ${height}`)

  const g = svg.append("g").attr("transform", `translate(${width / 2},${height / 2})`)

  const tooltip = svgWrap.append("div")
    .style("position", "absolute")
    .style("top", "0").style("left", "0")
    .style("pointer-events", "none")
    .style("background", "rgba(26,60,94,0.92)")
    .style("color", "white")
    .style("padding", "6px 11px")
    .style("border-radius", "6px")
    .style("font-size", "12px")
    .style("line-height", "1.4")
    .style("display", "none")

  function fmtPct(v) {
    return (v / total * 100).toLocaleString("es-CO", { minimumFractionDigits: 1, maximumFractionDigits: 1 })
  }

  function colorEsClaro(hex) {
    const c = hex.replace("#", "")
    const r = parseInt(c.substring(0, 2), 16), g = parseInt(c.substring(2, 4), 16), b = parseInt(c.substring(4, 6), 16)
    return (0.299 * r + 0.587 * g + 0.114 * b) / 255 > 0.6
  }

  g.selectAll("path")
    .data(arcos)
    .join("path")
      .attr("d", arco)
      .attr("fill", d => colores.get(d.data.grupo) ?? "#999")
      .attr("stroke", "#fff")
      .attr("stroke-width", 1)
      .on("mouseover", function(event, d) {
        d3.select(this).attr("opacity", 0.82)
        tooltip.style("display", "block")
          .html(`<b>${d.data.grupo}</b><br>${d.data.cantidad.toLocaleString("es-CO")} (${fmtPct(d.data.cantidad)}&nbsp;%)`)
      })
      .on("mousemove", function(event) {
        const [mx, my] = d3.pointer(event, svgWrap.node())
        tooltip.style("left", (mx + 12) + "px").style("top", (my - 34) + "px")
      })
      .on("mouseout", function() {
        d3.select(this).attr("opacity", 1)
        tooltip.style("display", "none")
      })

  // Etiquetas directas solo en los segmentos que superan el 8 % (evita amontonar texto).
  g.selectAll("text")
    .data(arcos.filter(d => d.data.cantidad / total >= 0.08))
    .join("text")
      .attr("transform", d => `translate(${arco_etiqueta.centroid(d)})`)
      .attr("text-anchor", "middle")
      .attr("font-size", "12px")
      .attr("font-weight", 700)
      .attr("fill", d => colorEsClaro(colores.get(d.data.grupo) ?? "#999") ? "#333" : "#fff")
      .text(d => `${(d.data.cantidad / total * 100).toLocaleString("es-CO", { maximumFractionDigits: 0 })}%`)

  svg.append("text")
    .attr("x", width / 2).attr("y", height / 2 - 4)
    .attr("text-anchor", "middle")
    .attr("font-size", "22px")
    .attr("font-weight", 700)
    .attr("fill", "#00514f")
    .text(total.toLocaleString("es-CO"))

  svg.append("text")
    .attr("x", width / 2).attr("y", height / 2 + 16)
    .attr("text-anchor", "middle")
    .attr("font-size", "11px")
    .attr("fill", "#666")
    .text("registros")

  const leyenda = container.append("div")
    .style("display", "flex")
    .style("flex-direction", "column")
    .style("gap", "6px")
    .style("font-size", "0.82rem")
    .style("flex", "1 1 200px")
    .style("min-width", "0")
    .style("max-width", "260px")

  leyenda.selectAll("div")
    .data(data)
    .join("div")
      .style("display", "flex")
      .style("align-items", "flex-start")
      .style("gap", "8px")
      .style("min-width", "0")
      .html(d => `
        <span style="display:inline-block;width:12px;height:12px;margin-top:3px;border-radius:3px;background:${colores.get(d.grupo) ?? "#999"};flex-shrink:0"></span>
        <span style="min-width:0">${d.grupo} — <b>${d.cantidad.toLocaleString("es-CO")}</b> (${fmtPct(d.cantidad)}&nbsp;%)</span>
      `)

  return container.node()
}
